## Survivorship / bias notes
- Both universes are today's large, surviving ETFs. They were chosen with hindsight, but they are broad index funds, which is a much smaller survivorship risk than single stocks.
- EFA starts in 2001 and EEM in 2003. XLRE and XLC are excluded because their history is too short. Yahoo's adjusted closes include dividends.
- The parameters match the well-known Connors (2008) and Faber/Antonacci momentum rules. They were not tuned here, and the grid shows no cliff around them.
- The 2x-cost test stands in for slippage. These are liquid ETFs at the open.
## strategies/rsi2_etf.py
```python
#!/usr/bin/env python3
"""
equity_etf_rsi2_dip_200d  --  RSI(2) dip-buy on broad equity ETFs with a 200-day trend filter.
Self-contained: needs only pandas, numpy, yfinance. No secrets, no network besides Yahoo.

RULES (exact)
  Universe   : SPY QQQ IWM EFA EEM (daily, split/dividend-adjusted closes, Yahoo auto_adjust)
  Indicators : RSI(2) Wilder smoothing (ewm alpha=1/2) on close; SMA200 and SMA5 of close
  Entry      : at close of day t, if RSI2 < RSI_LO (10) AND close > SMA200 (trend filter)
               -> BUY at the NEXT day's open. Long only. One position per symbol max.
  Exit       : first close where close > SMA5 -> SELL at the next open
               OR time stop: MAX_HOLD trading days after entry (default 10, a safety net;
               the SMA5 exit fires first in ~97% of trades, avg hold ~3.9 days)
               OR (optional) disaster stop SL_PCT below entry checked on close (default off
               in the backtest; the emitted pick carries an 8% catastrophe stop for the dashboard)
  TP         : no fixed take-profit; the "tp" field emitted = today's SMA5 (the level whose
               cross triggers the exit), for the dashboard's resolver only.
  Sizing     : equal weight, 1/5 of capital per symbol slot (cash when no signal). No leverage.
  Costs      : 3 bps per side (ETF), charged on every weight change at the open.
"""
from __future__ import annotations
import sys, json
from datetime import datetime, timezone
import numpy as np, pandas as pd

ALGO = "equity_etf_rsi2_dip_200d"
UNIVERSE = ["SPY", "QQQ", "IWM", "EFA", "EEM"]
RSI_LO, TREND, EXIT_SMA, MAX_HOLD, COST_BPS = 10, 200, 5, 10, 3.0
CATASTROPHE_SL = 0.08

def load(symbols=UNIVERSE, start="2000-01-01"):
    import yfinance as yf
    raw = yf.download(symbols, start=start, auto_adjust=True, progress=False)
    return raw["Open"][symbols], raw["Close"][symbols]

def rsi(c: pd.Series, n=2):
    d = c.diff(); up = d.clip(lower=0).ewm(alpha=1/n, adjust=False).mean()
    dn = (-d.clip(upper=0)).ewm(alpha=1/n, adjust=False).mean()
    return 100 - 100 / (1 + up / dn.replace(0, np.nan))

def positions(C: pd.DataFrame, rsi_lo=RSI_LO, trend=TREND, max_hold=MAX_HOLD, sl_pct=None, O=None):
    """Returns target position (0/1) decided at each close; executed next open."""
    P = pd.DataFrame(0.0, index=C.index, columns=C.columns)
    for s in C.columns:
        c = C[s].dropna()
        if len(c) < trend + 5: continue
        r, st, ex = rsi(c), c.rolling(trend).mean(), c.rolling(EXIT_SMA).mean()
        o = (O[s].reindex(c.index) if O is not None else c)
        pos = np.zeros(len(c)); inpos = False; held = 0; ep = None
        cv, rv, sv, xv = c.values, r.values, st.values, ex.values
        for i in range(len(c)):
            if inpos:
                held += 1
                if ep is None and i > 0: ep = o.values[i]       # filled at this open
                stop = sl_pct is not None and ep is not None and cv[i] < ep * (1 - sl_pct)
                if cv[i] > xv[i] or held >= max_hold or stop:
                    inpos = False; ep = None
            elif rv[i] < rsi_lo and cv[i] > sv[i]:
                inpos = True; held = 0; ep = None
            pos[i] = 1.0 if inpos else 0.0
        P.loc[c.index, s] = pos
    return P

def simulate(O, C, P, cost_bps=COST_BPS, weight=None):
    """P = target at close t; live from open t+1. Returns (daily portfolio return, trades df)."""
    weight = weight if weight is not None else 1.0 / C.shape[1]
    W = P.shift(1).fillna(0) * weight                   # weight held during day t (from open t)
    Wprev = W.shift(1).fillna(0)
    on = (O / C.shift(1) - 1).fillna(0); intra = (C / O - 1).fillna(0)
    cost = (W - Wprev).abs() * cost_bps / 1e4
    daily = (Wprev * on + W * intra - cost).sum(axis=1)
    trades = []
    for s in C.columns:
        w = W[s].values; idx = C.index
        i = 0
        while i < len(w):
            if w[i] > 0:
                j = i
                while j < len(w) and w[j] > 0: j += 1
                if j >= len(w): break                    # still open at end -> not counted
                e, x = O[s].iloc[i], O[s].iloc[j]
                if pd.notna(e) and pd.notna(x) and e > 0:
                    trades.append(dict(symbol=s, entry_date=idx[i], exit_date=idx[j], bars=j - i,
                                       ret=(x / e - 1) - 2 * cost_bps / 1e4))
                i = j
            else: i += 1
    return daily, pd.DataFrame(trades)

def backtest(O=None, C=None, **kw):
    if C is None: O, C = load()
    cost = kw.pop("cost_bps", COST_BPS)
    P = positions(C, O=O, **kw)
    return simulate(O, C, P, cost)

def signals(O=None, C=None, now=None):
    """Today's picks in the repo pick format (BUY = enter at next open)."""
    if C is None: O, C = load(start="2024-01-01")
    now = now or datetime.now(timezone.utc).isoformat()
    picks, status = [], {}
    for s in C.columns:
        c = C[s].dropna(); r = float(rsi(c).iloc[-1]); sma200 = float(c.rolling(TREND).mean().iloc[-1])
        sma5 = float(c.rolling(EXIT_SMA).mean().iloc[-1]); px = float(c.iloc[-1])
        status[s] = dict(date=str(c.index[-1].date()), close=round(px, 2), rsi2=round(r, 2),
                         sma200=round(sma200, 2), above_200d=px > sma200, sma5=round(sma5, 2),
                         entry_signal=bool(r < RSI_LO and px > sma200))
        if status[s]["entry_signal"]:
            picks.append(dict(symbol=s, direction="LONG", signal_type="BUY", entry_price=round(px, 4),
                take_profit=round(max(sma5, px * 1.001), 4), stop_loss=round(px * (1 - CATASTROPHE_SL), 4),
                max_hold_bars=MAX_HOLD, max_hold_hours=MAX_HOLD * 24 * 7 // 5,
                strategy=ALGO, algorithm=ALGO, source_system=ALGO, timestamp=now,
                asset_class="ETF", category="etf", timeframe="1d", confidence=0.6,
                reason=f"RSI2={r:.1f}<{RSI_LO}, close {px:.2f}>SMA200 {sma200:.2f}; exit on close>SMA5 or {MAX_HOLD}d",
                extra=dict(exit_rule="close>SMA5 -> next open", sizing="1/5 equity per slot", cost_bps_side=COST_BPS)))
    return picks, status

if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "backtest":
        d, t = backtest()
        g, l = t.ret[t.ret > 0].sum(), -t.ret[t.ret < 0].sum()
        print(f"n={len(t)} PF={g/l:.2f} CAGR={((1+d).prod()**(252/len(d))-1)*100:.2f}%")
    else:
        p, st = signals(); print(json.dumps(dict(status=st, picks=p), indent=1, default=str))
```

## strategies/sector_rotation.py
```python
#!/usr/bin/env python3
"""
equity_sector_rotation_12m_top3  --  monthly sector momentum rotation with SPY 200-day filter.
Self-contained: pandas, numpy, yfinance only. No secrets.

RULES (exact)
  Universe   : 9 original SPDR sectors XLK XLF XLE XLV XLY XLP XLI XLU XLB (+ SPY as the filter)
               (XLRE 2015 / XLC 2018 excluded: too short for the 2000+ test)
  Score      : 12-1 momentum = close[t-21] / close[t-252] - 1  (12 months, skip latest month)
  Rebalance  : last trading day of each month, at the close -> trades at the NEXT open
  Pick       : top K=3 sectors by score, equal weight 1/3 each
  Filter     : if SPY close <= SPY SMA200 at the rebalance close -> hold cash (all 0) that month
  Exit       : a sector is sold at the next open after a month-end where it is no longer top-3
               (or the SPY filter turns off). No fixed TP/SL in the backtest.
               Emitted pick: max_hold = 23 trading days (re-issued monthly if still top-3),
               tp = +15% / sl = -10% as wide dashboard bounds only (not part of the tested rules).
  Sizing     : 1/3 capital per holding, no leverage. Costs: 3 bps per side on weight changes.
"""
from __future__ import annotations
import sys, json
from datetime import datetime, timezone
import numpy as np, pandas as pd

ALGO = "equity_sector_rotation_12m_top3"
SECTORS = ["XLK", "XLF", "XLE", "XLV", "XLY", "XLP", "XLI", "XLU", "XLB"]
LOOKBACK, SKIP, TOP_K, FILTER_SMA, COST_BPS = 252, 21, 3, 200, 3.0

def load(start="2000-01-01"):
    import yfinance as yf
    syms = SECTORS + ["SPY"]
    raw = yf.download(syms, start=start, auto_adjust=True, progress=False)
    return raw["Open"][syms], raw["Close"][syms]

def month_end_mask(idx):
    s = pd.Series(idx, index=idx)
    return s.groupby([idx.year, idx.month]).transform("max") == s

def scores(C, lookback=LOOKBACK, skip=SKIP):
    return C[SECTORS].shift(skip) / C[SECTORS].shift(lookback) - 1

def positions(C, lookback=LOOKBACK, skip=SKIP, k=TOP_K, use_filter=True):
    sc = scores(C, lookback, skip); spy = C["SPY"]; ok = spy > spy.rolling(FILTER_SMA).mean()
    me = month_end_mask(C.index).values
    P = pd.DataFrame(np.nan, index=C.index, columns=SECTORS)
    for i, d in enumerate(C.index):
        if not me[i]: continue
        row = sc.loc[d].dropna()
        p = pd.Series(0.0, index=SECTORS)
        if len(row) >= k and (ok.loc[d] or not use_filter):
            p[row.nlargest(k).index] = 1.0
        P.loc[d] = p
    return P.ffill().fillna(0.0)

def simulate(O, C, P, cost_bps=COST_BPS, k=TOP_K):
    O, C = O[SECTORS], C[SECTORS]
    W = P.shift(1).fillna(0) / k; Wprev = W.shift(1).fillna(0)
    on = (O / C.shift(1) - 1).fillna(0); intra = (C / O - 1).fillna(0)
    daily = (Wprev * on + W * intra - (W - Wprev).abs() * cost_bps / 1e4).sum(axis=1)
    trades = []
    for s in SECTORS:
        w = W[s].values; i = 0
        while i < len(w):
            if w[i] > 0:
                j = i
                while j < len(w) and w[j] > 0: j += 1
                if j >= len(w): break
                e, x = O[s].iloc[i], O[s].iloc[j]
                trades.append(dict(symbol=s, entry_date=C.index[i], exit_date=C.index[j], bars=j - i,
                                   ret=(x / e - 1) - 2 * cost_bps / 1e4))
                i = j
            else: i += 1
    return daily, pd.DataFrame(trades)

def backtest(O=None, C=None, cost_bps=COST_BPS, **kw):
    if C is None: O, C = load()
    P = positions(C, **kw)
    return simulate(O, C, P, cost_bps, kw.get("k", TOP_K))

def signals(O=None, C=None, now=None):
    if C is None: O, C = load(start="2023-06-01")
    now = now or datetime.now(timezone.utc).isoformat()
    sc = scores(C).iloc[-1].sort_values(ascending=False)
    spy = C["SPY"]; spy_ok = bool(spy.iloc[-1] > spy.rolling(FILTER_SMA).mean().iloc[-1])
    ret12 = (C[SECTORS].iloc[-1] / C[SECTORS].iloc[-253] - 1)
    status = dict(date=str(C.index[-1].date()), spy_above_200d=spy_ok,
                  ranking=[dict(symbol=s, mom_12_1_pct=round(v * 100, 2), ret_12m_pct=round(ret12[s] * 100, 2))
                           for s, v in sc.items()])
    picks = []
    if spy_ok:
        for rank, s in enumerate(sc.index[:TOP_K], 1):
            px = float(C[s].iloc[-1])
            picks.append(dict(symbol=s, direction="LONG", signal_type="BUY", entry_price=round(px, 4),
                take_profit=round(px * 1.15, 4), stop_loss=round(px * 0.90, 4),
                max_hold_bars=23, max_hold_hours=23 * 24 * 7 // 5, strategy=ALGO, algorithm=ALGO,
                source_system=ALGO, timestamp=now, asset_class="ETF", category="etf", timeframe="1mo",
                confidence=0.6, reason=f"rank {rank}/9 by 12-1 momentum {sc[s]*100:.1f}%; SPY>SMA200",
                extra=dict(rebalance="month-end close -> next open", target_weight=round(1 / TOP_K, 4),
                           note="hold while in top-3 at month end; tp/sl are wide dashboard bounds only")))
    return picks, status

if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "backtest":
        d, t = backtest(); g, l = t.ret[t.ret > 0].sum(), -t.ret[t.ret < 0].sum()
        print(f"n={len(t)} PF={g/l:.2f} CAGR={((1+d).prod()**(252/len(d))-1)*100:.2f}%")
    else:
        p, st = signals(); print(json.dumps(dict(status=st, picks=p), indent=1, default=str))
```

```python
import pandas as pd, numpy as np, json, itertools
from scipy.stats import norm, skew, kurtosis
import rsi2_etf as R, sector_rotation as S
rng = np.random.default_rng(7)
r = pd.read_pickle('px.pkl'); O, C = r['Open'], r['Close']
OR, CR = O[R.UNIVERSE], C[R.UNIVERSE]
def pf(x): g, l = x[x > 0].sum(), -x[x < 0].sum(); return g / l if l > 0 else np.inf
def mdd(d): e = (1 + d).cumprod(); return (e / e.cummax() - 1).min() * 100
def sharpe(d): d = d[d.index >= d.ne(0).idxmax()]; return d.mean() / d.std()
def boot(t, B=2000):
    g = t.groupby(t.entry_date.dt.normalize())['ret'].apply(list).tolist(); out = []
    for _ in range(B):
        s = [x for i in rng.integers(0, len(g), len(g)) for x in g[i]]; out.append(pf(np.array(s)))
    return np.percentile(out, 5)
def mc(t, Oo, cost, runs=1000):
    real = pf(t.ret.values); arrs = {s: Oo[s].dropna() for s in t.symbol.unique()}; res = []
    syms, bars = t.symbol.values, t.bars.values
    for _ in range(runs):
        rr = np.empty(len(t))
        for k, (s, b) in enumerate(zip(syms, bars)):
            a = arrs[s].values; i = rng.integers(0, len(a) - b); rr[k] = a[i + b] / a[i] - 1 - 2 * cost / 1e4
        res.append(pf(rr))
    return (np.array(res) < real).mean() * 100
def full(name, d, t, Oo, cost, d2x_t, spyok):
    d = d[d.index >= d.ne(0).idxmax()]
    x = t.ret.values; gp = t[t.ret > 0].groupby('symbol').ret.sum(); top = gp.idxmax()
    n = len(t); half = t.entry_date.sort_values().iloc[n // 2]
    roll = []
    for y in range(2000, 2023):
        w = t[(t.entry_date.dt.year >= y) & (t.entry_date.dt.year < y + 5)]
        if len(w) >= 10: roll.append(pf(w.ret.values))
    ent = t.entry_date
    reg = {'bull(SPY>200d)': pf(t[spyok.reindex(ent).values].ret.values),
           'bear(SPY<=200d)': (lambda z: (len(z), pf(z.ret.values)))(t[~spyok.reindex(ent).values.astype(bool)]),
           '2000-2009': pf(t[ent.dt.year < 2010].ret.values), '2010-2019': pf(t[(ent.dt.year >= 2010) & (ent.dt.year < 2020)].ret.values),
           '2020-2026': pf(t[ent.dt.year >= 2020].ret.values)}
    return dict(name=name, n=n, PF=pf(x), PF_CI_lo=boot(t), WR=(x > 0).mean() * 100 if n >= 20 else None,
        CAGR=((1 + d).prod() ** (252 / len(d)) - 1) * 100, MDD=mdd(d), Sharpe_ann=sharpe(d) * np.sqrt(252),
        MC_pct=mc(t, Oo, cost), PF_2x=pf(d2x_t.ret.values), top_sym=top, top_share=gp.max() / gp.sum() * 100,
        PF_ex_top=pf(t[t.symbol != top].ret.values), PF_H1=pf(t[t.entry_date < half].ret.values),
```
